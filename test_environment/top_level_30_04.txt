// =============================================================
//  top_level.sv  –  updated to match actual module port names
// =============================================================
`timescale 1ns/1ps

module top_level (
    input  wire         MAX10_CLK1_50,
    input  wire         KEY_0,

    input  wire [15:0]  FPGA_DATA_IN,
    output wire [15:0]  FPGA_DATA_OUT,
    output wire         FPGA_OUTPUT_READY,

    output wire [12:0]  DRAM_ADDR,
    output wire [1:0]   DRAM_BA,
    output wire         DRAM_CAS_N,
    output wire         DRAM_CKE,
    output wire         DRAM_CLK,
    output wire         DRAM_CS_N,
    inout  wire [15:0]  DRAM_DQ,
    output wire         DRAM_LDQM,
    output wire         DRAM_RAS_N,
    output wire         DRAM_UDQM,
    output wire         DRAM_WE_N,
    output wire [9:0]   LEDR
);

    tri [15:0] bus;

    wire clk_sys = MAX10_CLK1_50;
    wire rst_n;
    wire start_pulse;

    wire debounced_key;
    debounce_button #(.DELAY_MAX(100_000)) u_deb (
        .clk       (clk_sys),
        .rst_n     (1'b1),
        .noisy_in  (~KEY_0),
        .clean_out (debounced_key)
    );

    reset_and_start u_rst (
        .clk          (clk_sys),
        .db_button_in (debounced_key),
        .reset_n_out  (rst_n),
        .start_pulse  (start_pulse)
    );

    wire [23:0] sdram_addr;
    wire [15:0] sdram_din, sdram_dout;
    wire sdram_rd, sdram_wr, sdram_valid;

    SDRAM_CONTROLLER_wrapper u_sdram (
        .clk         (clk_sys),
        .reset_n     (rst_n),
        .address     (sdram_addr),
        .data_in     (sdram_din),
        .read_en     (sdram_rd),
        .data_out    (sdram_dout),
        .initialized (),
        .sdram_addr  (DRAM_ADDR),
        .sdram_ba    (DRAM_BA),
        .sdram_cke   (DRAM_CKE),
        .sdram_cs_n  (DRAM_CS_N),
        .sdram_ras_n (DRAM_RAS_N),
        .sdram_cas_n (DRAM_CAS_N),
        .sdram_we_n  (DRAM_WE_N),
        .sdram_ldqm  (DRAM_LDQM),
        .sdram_udqm  (DRAM_UDQM),
        .sdram_dq    (DRAM_DQ)
    );

    wire [10:0] ram_addr;
    wire [15:0] ram_din, ram_dout;
    wire ram_wr, ram_rd;

    on_chip_memory u_ram (
        .clk              (clk_sys),
        .reset_n          (rst_n),
        .wr_en            (ram_wr),
        .rd_en            (ram_rd),
        .data_in          (ram_din),
        .data_out         (ram_dout),
        .multi_cycle_mode (1'b0),
        .cycle_count      (2'd1),
        .address_in       (ram_addr),
        .use_external_addr(1'b1)
    );

    wire ctrl_output_ready, ctrl_start_run;
    wire ctrl_wr_en;
    wire [9:0] ctrl_mem_address;
    wire led_done_wire;

   control_unit #(.LOAD_DEPTH(69)) ctrl (
        .clk           (clk_internal),
        .reset_n       (reset_n_sys),
        .start         (start_sig),
        .sdram_rd_en   (sdram_rd_en),
        .sdram_wr_en   (sdram_wr_en),
        .sdram_address (sdram_address),
        .sdram_dout    (sdram_data_out),
        .sdram_ready   (sdram_ready),
        .mem_wr_en     (ctrl_wr_en),
        .mem_address   (ctrl_mem_address),
        .output_ready  (ctrl_output_ready),
        .start_run     (ctrl_start_run),
        .all_done      (ctrl_all_done),
        .led_done      (led_done_wire)
    );

    wire [10:0] gen_address;
    wire        gen_rd_en, gen_wr_en, gen_chip_sel;
    wire [15:0] gen_data_out;

    test_generator u_gen (
        .clk         (clk_sys),
        .reset       (~rst_n),
        .start       (ctrl_start_run),
        .address_BUS (gen_address),
        .DATA_BUS    (gen_data_out),
        .rd_en       (gen_rd_en),
        .wr_en       (gen_wr_en),
        .output_ready(output_ready),
        .chip_sel    (gen_chip_sel)
    );

    wire [15:0] mac_data_out;
    wire        mac_ready;
    wire        mac_binary_out;

    top u_dut (
        .clk(clk_sys),
        .bus(bus),
        .wr_en(gen_wr_en),
        .rd_en(gen_rd_en),
        .chip_sel(gen_chip_sel),
        .output_ready(mac_ready),
        .output_bit(mac_binary_out),
        .wr_data_ptr(),
        .rd_data_ptr(),
        .ctrl_state()
    );

    assign bus = (gen_wr_en && gen_chip_sel) ? ram_dout : 16'bz;
    wire [15:0] data_from_dut = bus;

    wire [10:0] val_address_out;
    wire        val_rd_en, val_wr_en;
    wire [15:0] val_data_to_mem;
    wire        val_done;

    validator u_val (
        .clk           (clk_sys),
        .reset_n       (rst_n),
        .dut_data_out  (bus),
        .dut_single_out(mac_binary_out),
        .output_ready  (mac_ready),
        .address_out   (val_address_out),
        .rd_en         (val_rd_en),
        .wr_en         (val_wr_en),
        .mem_data_out  (ram_dout),
        .data_to_mem   (val_data_to_mem),
        .gen_wr_en     (gen_wr_en),
        .val_done      (val_done)
    );

    reg [10:0] mux_address;
    reg [15:0] mux_data_in_r;
    reg        mux_wr_en, mux_rd_en_r;

    always @(*) begin
        mux_wr_en     = 1'b0;
        mux_rd_en_r   = 1'b0;
        mux_address   = 11'd0;
        mux_data_in_r = 16'd0;

        if (ctrl_wr_en) begin
            mux_wr_en     = 1'b1;
            mux_address   = ctrl_mem_address;
            mux_data_in_r = sdram_dout;
        end else if (gen_wr_en || gen_rd_en) begin
            mux_wr_en     = gen_wr_en;
            mux_rd_en_r   = gen_rd_en;
            mux_address   = gen_address;
            mux_data_in_r = 16'd0;
        end else if (val_wr_en || val_rd_en) begin
            mux_wr_en     = val_wr_en;
            mux_rd_en_r   = val_rd_en;
            mux_address   = val_address_out;
            mux_data_in_r = val_data_to_mem;
        end
    end

    assign ram_wr     = mux_wr_en;
    assign ram_rd     = mux_rd_en_r;
    assign ram_addr   = mux_address;
    assign ram_din    = mux_data_in_r;

    wire any_ready = mac_ready || val_done || ctrl_output_ready;
    assign FPGA_OUTPUT_READY = any_ready;

    reg [15:0] internal_data;
    always @(*) begin
        if (val_done)
            internal_data = 16'hF0F0;
        else if (mac_ready)
            internal_data = mac_data_out;
        else if (ctrl_output_ready)
            internal_data = 16'h9999;
        else
            internal_data = 16'h0000;
    end

    assign DRAM_CLK = clk_sys;
    assign FPGA_DATA_OUT = internal_data;
    assign LEDR[9] = led_done_wire;
    assign LEDR[8:0] = 9'b0;

endmodule
